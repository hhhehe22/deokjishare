function hashPassword_(pw, salt) {
 const bytes = Utilities.computeDigest(
   Utilities.DigestAlgorithm.SHA_256, salt + ':' + pw, Utilities.Charset.UTF_8
 );
 return Utilities.base64Encode(bytes);
}
function safeEquals_(a, b) {
 a = String(a); b = String(b);
 if (a.length !== b.length) return false;
 let diff = 0;
 for (let i = 0; i < a.length; i++) diff |= (a.charCodeAt(i) ^ b.charCodeAt(i));
 return diff === 0;
}
function signToken_(payload) {
 const secret = PropertiesService.getScriptProperties().getProperty(PROP_SECRET);
 if (!secret) return '';
 return Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(payload, secret));
}
function makeToken_() {
 const payload = String(Date.now() + TEACHER_TOKEN_TTL_MS);
 return payload + '.' + signToken_(payload);
}
function isValidToken_(token) {
 if (!token || typeof token !== 'string') return false;
 const parts = token.split('.');
 if (parts.length !== 2) return false;
 const exp = Number(parts[0]);
 if (!exp || exp < Date.now()) return false;
 const expected = signToken_(parts[0]);
 if (!expected) return false;
 return safeEquals_(expected, parts[1]);
}

function setupTeacherPassword(pw) {
 const lock = LockService.getScriptLock();
 try {
   lock.waitLock(5000);
   if (isPasswordSet_()) {
     return { success: false, message: '이미 교사 비밀번호가 설정되어 있습니다.' };
   }
   const clean = String(pw || '').trim();
   if (clean.length < TEACHER_MIN_PW_LENGTH) {
     return { success: false, message: '비밀번호는 ' + TEACHER_MIN_PW_LENGTH + '자 이상으로 설정해주세요.' };
   }
   const salt = Utilities.getUuid();
   PropertiesService.getScriptProperties().setProperties({
     [PROP_PW_SALT]: salt,
     [PROP_PW_HASH]: hashPassword_(clean, salt),
     [PROP_SECRET]: Utilities.getUuid() + Utilities.getUuid()
   });
   return { success: true, token: makeToken_() };
 } catch (err) {
   return { success: false, message: '잠시 후 다시 시도해주세요.' };
 } finally {
   try { lock.releaseLock(); } catch (e) {}
 }
}
function loginTeacher(pw) {
 if (!isPasswordSet_()) {
   return { success: false, message: '아직 교사 비밀번호가 설정되지 않았습니다.' };
 }
 const cache = CacheService.getScriptCache();
 const failKey = 'teacher_login_fails';
 const fails = Number(cache.get(failKey) || 0);
 if (fails >= LOGIN_MAX_FAILS) {
   return { success: false, message: '비밀번호를 여러 번 틀려 잠시 잠겼습니다. 5분 뒤 다시 시도해주세요.' };
 }
 const props = PropertiesService.getScriptProperties();
 const salt = props.getProperty(PROP_PW_SALT) || '';
 const savedHash = props.getProperty(PROP_PW_HASH) || '';
 const inputHash = hashPassword_(String(pw || '').trim(), salt);
 if (!safeEquals_(inputHash, savedHash)) {
   cache.put(failKey, String(fails + 1), LOGIN_LOCK_SECONDS);
   return { success: false, message: '비밀번호가 일치하지 않습니다.' };
 }
 cache.remove(failKey);
 return { success: true, token: makeToken_() };
}

function resolveTargetFolder(folderInput, boardTitle) {
 let input = (folderInput || '').trim();
 if (input !== '') {
   const match = input.match(/folders\/([a-zA-Z0-9-_]+)/);
   if (match && match[1]) return match[1];
   const matchId = input.match(/id=([a-zA-Z0-9-_]+)/);
   if (matchId && matchId[1]) return matchId[1];
   if (!input.includes('/') && input.length > 15) return input;
 }
 try {
   const ssFile = DriveApp.getFileById(SpreadsheetApp.getActiveSpreadsheet().getId());
   const parentFolders = ssFile.getParents();
   let parentFolder = parentFolders.hasNext() ? parentFolders.next() : DriveApp.getRootFolder();
   const newFolderName = `[${boardTitle}] 업로드자료`;
   const createdFolder = parentFolder.createFolder(newFolderName);
   try {
     createdFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
   } catch (e) {}
   return createdFolder.getId();
 } catch (err) {
   return '';
 }
}
function getPosts(sheetName) {
 const ss = SpreadsheetApp.getActiveSpreadsheet();
 const sheet = ss.getSheetByName(sheetName);
 if (!sheet) return [];
 const data = sheet.getDataRange().getValues();
 if (data.length <= 1) return [];
 const posts = data.slice(1).map(r => {
   let comments = [];
   try {
     if (r[10]) comments = JSON.parse(r[10]);
   } catch (e) {
     comments = [];
   }
   const status = r[11] ? String(r[11]) : 'approved';
   return {
     id: String(r[0]),
     category: String(r[1] || '전체'),
     title: cellToText_(r[2]),
     content: cellToText_(r[3]),
     author: cellToText_(r[4]),
     color: r[6] || '#ffffff',
     imageUrl: r[7] || '',
     likes: Number(r[8]) || 0,
     timestamp: r[9] ? Utilities.formatDate(new Date(r[9]), "Asia/Seoul", "MM/dd HH:mm") : '',
     comments: comments,
     status: status,
     order: parseOrder_(r[12]),
     pinnedAt: parsePinnedAt_(r[13])
   };
 }).map(p => {
   p.pinned = p.pinnedAt > 0;
   return p;
 }).reverse().reduce((acc, p) => {
   // 교사가 순서를 정한 글(order 있음)은 번호순, 아직 정하지 않은 새 글은 맨 위(최신순)
   (p.order === null ? acc.free : acc.fixed).push(p);
   return acc;
 }, { free: [], fixed: [] });
 const ordered = posts.free.concat(posts.fixed.sort((a, b) => a.order - b.order));
 // 교사가 고정한 글은 항상 맨 앞(가장 최근에 고정한 글이 제일 위). 선반 화면에서는 각 선반의 맨 위가 된다.
 const pinned = ordered.filter(p => p.pinned).sort((a, b) => b.pinnedAt - a.pinnedAt);
 const normal = ordered.filter(p => !p.pinned);
 return pinned.concat(normal);
}
function parsePinnedAt_(v) {
 const n = Number(v);
 return (isFinite(n) && n > 0) ? n : 0;
}
function parseOrder_(v) {
 if (v === '' || v === null || v === undefined) return null;
 const n = Number(v);
 return isFinite(n) ? n : null;
}

    function rgbToHex(col) {
      if (!col) return '';
      if (col.startsWith('#')) return col.toLowerCase();
      const rgb = col.match(/\d+/g);
      if (!rgb || rgb.length < 3) return col.toLowerCase();
      return '#' + ((1 << 24) + (Number(rgb[0]) << 16) + (Number(rgb[1]) << 8) + Number(rgb[2])).toString(16).slice(1).toLowerCase();
    }

    function initSortable() {
      destroySortables();
      const mainEl = document.getElementById('mainContainer');
      mainEl.classList.toggle('sortable-enabled', canDragNow());
      if (!canDragNow()) return;
      const lists = Array.from(mainEl.querySelectorAll('.shelf-posts, .grid-wrap'));
      lists.forEach(listEl => {
        sortableInstances.push(new Sortable(listEl, {
          group: 'dukji-posts',
          draggable: '.post-card',
          animation: 150,
          ghostClass: 'drag-ghost',
          chosenClass: 'drag-chosen',
          filter: '.post-tools, button, a, .post-card.pinned',   // 버튼은 그대로 클릭 · 고정 카드는 끌지 않음
          preventOnFilter: false,
          delay: 150,                          // 터치 화면에서 스크롤과 구분
          delayOnTouchOnly: true,
          touchStartThreshold: 4,
          onMove: (evt) => {
            // 고정 카드 위·사이에는 놓을 수 없고, 마지막 고정 카드 바로 뒤(일반 글 맨 위)까지만 허용
            const rel = evt.related;
            if (!rel || !rel.classList || !rel.classList.contains('pinned')) return true;
            const next = rel.nextElementSibling;
            const nextIsPinned = !!(next && next.classList && next.classList.contains('pinned'));
            return !!evt.willInsertAfter && !nextIsPinned;
          },
          onStart: () => { isDraggingNow = true; },
          onEnd: handleSortEnd
        }));
      });
    }
    function listCategory(listEl) {
      if (currentBoard.layout !== 'shelf') return '전체';

    function listPostIds(listEl) {
      return Array.from(listEl.children)
        .filter(el => el.classList && el.classList.contains('post-card') && el.getAttribute('data-id'))
        .map(el => el.getAttribute('data-id'));
    }
    function handleSortEnd(evt) {
      isDraggingNow = false;
      if (!isTeacher || !currentBoard) return;
      const from = evt.from, to = evt.to;
      if (from === to && evt.oldIndex === evt.newIndex) return;
      const lists = (from === to) ? [to] : [from, to];
      const items = [];
      lists.forEach(listEl => {
        const cat = listCategory(listEl);
        if (cat === undefined) return;
        listPostIds(listEl).forEach((id, i) => items.push({ id: id, category: cat, order: i + 1 }));
      });
      if (items.length === 0) return;
      // 화면 데이터에 먼저 반영(즉시 반응) → 서버에 저장
      const byId = {};
      items.forEach(it => { byId[it.id] = it; });
      currentPosts.forEach(p => {
        const it = byId[p.id];
        if (it) { p.category = it.category; p.order = it.order; }
      });
      currentPosts = orderPosts(currentPosts);
      setTimeout(drawBoardPosts, 0);
      pendingLayoutSaves++;
      google.script.run
        .withSuccessHandler(res => {
          pendingLayoutSaves = Math.max(0, pendingLayoutSaves - 1);
          if (handleAuthFail(res)) return;
          if (res && res.success) {
            showToast('순서를 저장했어요.');
          } else {
            showToast((res && res.message) ? res.message : '순서 저장에 실패했어요. 화면을 새로 불러옵니다.');
            fetchBoardManual();
          }
        })
        .withFailureHandler(() => {
          pendingLayoutSaves = Math.max(0, pendingLayoutSaves - 1);
          showToast('서버 통신에 실패했어요. 화면을 새로 불러옵니다.');
          fetchBoardManual();
        })
        .savePostLayout(currentBoard.sheetName, items, teacherToken);
    }
    // 서버(getPosts)와 같은 규칙으로 정렬: 고정 글(최근 고정 순) → 새 글(최신순) → 교사가 순서를 정한 글(번호순)
    function orderPosts(list) {
      const hasOrder = p => !(p.order === null || p.order === undefined);
      const pinned = list.filter(p => p.pinned).sort((a, b) => (b.pinnedAt || 0) - (a.pinnedAt || 0));
      const rest = list.filter(p => !p.pinned);
      const free = rest.filter(p => !hasOrder(p));
      const fixed = rest.filter(hasOrder).sort((a, b) => a.order - b.order);
      return pinned.concat(free, fixed);
    }

    function initPaste() {
      window.addEventListener('paste', e => {

    function compressImage(file) {
      if (!file || !file.type.startsWith('image/')) {
        showToast('이미지 파일만 선택할 수 있습니다.');
        return;
      }
      const submitBtn = document.getElementById('postSubmitBtn');
      if (submitBtn) {
        submitBtn.disabled = true;
        submitBtn.textContent = '사진 준비 중...';
      }
      const reader = new FileReader();
      reader.onload = function(e) {
        const img = new Image();
        img.onload = function() {
          const cvs = document.createElement('canvas');
          let w = img.width;
          let h = img.height;
          const maxDim = 700;
          if (w > maxDim || h > maxDim) {
            if (w > h) {
              h = Math.round((h * maxDim) / w);
              w = maxDim;
            } else {
              w = Math.round((w * maxDim) / h);
              h = maxDim;
            }
          }
          cvs.width = w;
          cvs.height = h;
          const ctx = cvs.getContext('2d');
          ctx.drawImage(img, 0, 0, w, h);
          attachedImgBase64 = cvs.toDataURL('image/jpeg', 0.65);
          document.getElementById('imgPreview').src = attachedImgBase64;
          document.getElementById('imgPreviewArea').style.display = 'block';
          if (submitBtn) {
            submitBtn.disabled = false;
            submitBtn.textContent = '올리기';
          }
          showToast('사진이 준비되었습니다.');
        };
        img.src = e.target.result;
      };
      reader.readAsDataURL(file);
    }

웹앱으로 배포하기 (가장 중요!)

최초 1회 권한 검토 및 승인 (경고창 해결)

최초 배포 시 구글 보안 경고창이 나타날 수 있습니다. 이는 비공인 개인 스크립트이므로 정상적인 과정입니다:

  1. [액세스 승인] 버튼 클릭
  2. 사용할 본인의 구글 계정 선택
  3. "Google에서 확인하지 않은 앱" 화면에서 좌측 하단 [고급 (Advanced)] 클릭
  4. [...으로 이동(안전하지 않음)] 링크 클릭 후 [허용] 완료

배포가 끝나면 발급되는 웹 앱 URL(https://script.google.com/macros/s/.../exec)을 구글 사이트에 임베드하거나 사용자에게 공유하면 됩니다!
